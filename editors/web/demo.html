LLLang syntax highlighting

Sample uses modern LLLang style: no semicolons, primary constructors, switch expressions, and string interpolation. 

namespace ShapesDemo

import Std.IO

public protocol Drawable {
    func draw() -> String
    func label() -> String {
        draw()
    }
}

public struct Circle(radius: Double): Drawable {
    public func draw() -> String {
        "circle r=\(radius)"
    }
}

public class Polygon(
    public let sides: Int,
    public let sideLength: Double
): Drawable {
    public virtual func draw() -> String {
        "polygon sides=\(sides) len=\(sideLength)"
    }
}

public func main() throws(DecodingError) {
    let shapes: [Drawable] = [
        Circle(2.5),
        Polygon(sides: 5, sideLength: 3.0),
    ]

    for shape in shapes {
        let msg = switch shape is Circle {
            true => "round: \(shape.label())",
            false => shape.label(),
        }
        print(msg)
    }

    /* async-friendly HTTP fetch (illustrative) */
    let body = await HTTP.get("https://api.example/v1/health")
    guard body.status == 200 else {
        throw NetworkError.unavailable
    }
}
